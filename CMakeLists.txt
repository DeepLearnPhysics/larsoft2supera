cmake_minimum_required(VERSION 3.20)
option(LARSOFT2SUPERA_BUILD_ART "Build against installed LArSoft and art" ON)
option(LARSOFT2SUPERA_BUILD_WRITER_MODULE "Build the art module with the LArCV2 output adapter" ON)
if(LARSOFT2SUPERA_BUILD_ART)
  find_package(cetmodules 3.20 REQUIRED)
endif()
option(LARSOFT2SUPERA_BUILD_GENERATOR "Build the ROOT/nusimdata generator adapter" ${LARSOFT2SUPERA_BUILD_ART})
project(larsoft2supera VERSION 0.1.0 LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(GNUInstallDirs)
find_package(Supera 2 CONFIG REQUIRED COMPONENTS data)

if(LARSOFT2SUPERA_BUILD_ART)
  include(CetCMakeEnv)
  cet_cmake_env()
  find_package(art REQUIRED)
  find_package(lardataobj REQUIRED)
  find_package(sbnobj REQUIRED)
  if(LARSOFT2SUPERA_BUILD_WRITER_MODULE)
    find_package(Supera 2 CONFIG REQUIRED COMPONENTS larcv2 processing)
  endif()
else()
  # Fast tests compile the genuine pinned value types, not replacements or mocks.
  # These source trees are supplied by the caller and are never installed.
  foreach(dependency IN ITEMS LARDATAOBJ_SOURCE SBNOBJ_SOURCE)
    if(NOT IS_DIRECTORY "${${dependency}}")
      message(FATAL_ERROR "Set ${dependency} to the pinned dependency source tree")
    endif()
  endforeach()
  add_library(test_value_types STATIC
    "${LARDATAOBJ_SOURCE}/lardataobj/RecoBase/OpFlash.cxx")
  target_include_directories(test_value_types SYSTEM PUBLIC
    "${LARDATAOBJ_SOURCE}")
  add_library(lardataobj::RecoBase ALIAS test_value_types)
  add_library(sbnobj_crt_types INTERFACE)
  target_include_directories(sbnobj_crt_types SYSTEM INTERFACE "${SBNOBJ_SOURCE}")
  add_library(sbnobj::Common_CRT ALIAS sbnobj_crt_types)
endif()

add_library(optical STATIC src/Optical.cxx)
add_library(larsoft2supera::optical ALIAS optical)
set_target_properties(optical PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(optical PUBLIC cxx_std_17)
target_compile_options(optical PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(optical PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(optical PUBLIC lardataobj::RecoBase supera::data)

add_library(auxiliary STATIC src/AuxDet.cxx)
add_library(larsoft2supera::auxiliary ALIAS auxiliary)
set_target_properties(auxiliary PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(auxiliary PUBLIC cxx_std_17)
target_compile_options(auxiliary PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(auxiliary PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(auxiliary PUBLIC sbnobj::Common_CRT supera::data)

if(LARSOFT2SUPERA_BUILD_ART)
  add_library(spacepoint STATIC src/SpacePoint.cxx)
  add_library(larsoft2supera::spacepoint ALIAS spacepoint)
  set_target_properties(spacepoint PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_features(spacepoint PUBLIC cxx_std_17)
  target_compile_options(spacepoint PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(spacepoint PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(spacepoint PUBLIC lardataobj::RecoBase canvas::canvas supera::data)
  install(TARGETS spacepoint EXPORT opticalTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  if(BUILD_TESTING)
    add_executable(spacepoint_test tests/SpacePoint_test.cxx)
    target_link_libraries(spacepoint_test PRIVATE spacepoint)
    target_compile_options(spacepoint_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME spacepoint_fields COMMAND spacepoint_test)
  endif()
endif()

if(LARSOFT2SUPERA_BUILD_GENERATOR)
  find_package(nusimdata REQUIRED)
  add_library(generator STATIC src/Generator.cxx)
  add_library(larsoft2supera::generator ALIAS generator)
  set_target_properties(generator PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_features(generator PUBLIC cxx_std_17)
  target_compile_options(generator PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_include_directories(generator PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(generator PUBLIC nusimdata::SimulationBase supera::data)
  if(BUILD_TESTING)
    add_executable(generator_test tests/Generator_test.cxx)
    target_link_libraries(generator_test PRIVATE generator)
    target_compile_options(generator_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME generator_fields COMMAND generator_test)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(optical_test tests/Optical_test.cxx)
  target_link_libraries(optical_test PRIVATE optical)
  target_compile_options(optical_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME optical_fields COMMAND optical_test)
  add_executable(auxiliary_test tests/AuxDet_test.cxx)
  target_link_libraries(auxiliary_test PRIVATE auxiliary)
  target_compile_options(auxiliary_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME auxiliary_fields COMMAND auxiliary_test)
endif()

if(LARSOFT2SUPERA_BUILD_ART)
  if(LARSOFT2SUPERA_BUILD_WRITER_MODULE)
    if(NOT LARSOFT2SUPERA_BUILD_GENERATOR)
      message(FATAL_ERROR "The composed writer module requires the generator adapter")
    endif()
    find_package(messagefacility REQUIRED)
    include(BuildPlugins)
    add_subdirectory(plugins)
  endif()
  install(TARGETS optical auxiliary EXPORT opticalTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  if(LARSOFT2SUPERA_BUILD_GENERATOR)
    install(TARGETS generator EXPORT opticalTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  endif()
  install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(DIRECTORY fcl/ DESTINATION ${CMAKE_INSTALL_DATADIR}/larsoft2supera/fcl)
  install(EXPORT opticalTargets NAMESPACE larsoft2supera::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/larsoft2supera)
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/larsoft2superaConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/larsoft2superaConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/larsoft2supera)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/larsoft2superaConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/larsoft2supera)
endif()
